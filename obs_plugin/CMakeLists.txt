cmake_minimum_required(VERSION 3.28)
project(obs_mate_menu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
option(BRIDGE_HOST_TEST_ONLY "Build the Windows native test host without OBS headers or libraries" OFF)
set(OBS_REFERENCE "" CACHE PATH "Prepared OBS 32.2.1 reference directory; see scripts/prepare_obs_sdk.ps1")
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This plugin and its native test host require Windows and an MSVC x64 toolchain.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Configure an x64 build. The supported OBS installation is 64-bit.")
endif()
# Keep project/header rebuilds correct even when localized MSVC /showIncludes
# output cannot be parsed by Ninja. Native Windows SDK/toolchain upgrades should
# still use a clean build directory.
set_property(SOURCE obs_mate_menu.cpp test_host.cpp APPEND PROPERTY OBJECT_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/bridge_policy.h;${CMAKE_CURRENT_SOURCE_DIR}/test_obs_api.h")
if(NOT BRIDGE_HOST_TEST_ONLY)
    if(NOT OBS_REFERENCE)
        message(FATAL_ERROR "Set -DOBS_REFERENCE=<prepared reference directory>, or use -DBRIDGE_HOST_TEST_ONLY=ON. Run scripts/prepare_obs_sdk.ps1 to prepare headers and import libraries.")
    endif()
    foreach(reference_file IN ITEMS obs-src/libobs/obs-module.h obs-src/frontend/api/obs-frontend-api.h generated/obsconfig.h generated/obs.lib generated/obs-frontend-api.lib)
        if(NOT EXISTS "${OBS_REFERENCE}/${reference_file}")
            message(FATAL_ERROR "Missing OBS reference file: ${OBS_REFERENCE}/${reference_file}. Run scripts/prepare_obs_sdk.ps1 first.")
        endif()
    endforeach()
    file(GLOB_RECURSE OBS_REFERENCE_HEADERS
        "${OBS_REFERENCE}/obs-src/libobs/*.h"
        "${OBS_REFERENCE}/obs-src/frontend/api/*.h")
    set_property(SOURCE obs_mate_menu.cpp APPEND PROPERTY OBJECT_DEPENDS
        "${OBS_REFERENCE_HEADERS};${OBS_REFERENCE}/generated/obsconfig.h")
    add_library(obs-mate-menu MODULE obs_mate_menu.cpp)
    target_include_directories(obs-mate-menu PRIVATE "${OBS_REFERENCE}/obs-src/libobs" "${OBS_REFERENCE}/obs-src/frontend/api" "${OBS_REFERENCE}/generated")
    target_compile_definitions(obs-mate-menu PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_compile_options(obs-mate-menu PRIVATE /W4 /permissive- /utf-8 /EHsc)
    target_link_libraries(obs-mate-menu PRIVATE "${OBS_REFERENCE}/generated/obs.lib" "${OBS_REFERENCE}/generated/obs-frontend-api.lib" advapi32 user32 shell32)
endif()
add_executable(bridge_plugin_test test_host.cpp obs_mate_menu.cpp)
target_compile_definitions(bridge_plugin_test PRIVATE BRIDGE_TEST_HOST WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(bridge_plugin_test PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(bridge_plugin_test PRIVATE advapi32 user32 shell32)
enable_testing()
add_test(NAME bridge_plugin_integration COMMAND bridge_plugin_test)
set_tests_properties(bridge_plugin_integration PROPERTIES TIMEOUT 30)
